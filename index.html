import React, { useState, useEffect, useRef } from "react";

import { Sparkles, MessageSquare, ChevronRight, Disc, Send, Radio } from "lucide-react";

export default function App() {

  const [level, setLevel] = useState(null);

  const [messages, setMessages] = useState([]);

  const [inputText, setInputText] = useState("");

  const [isTyping, setIsTyping] = useState(false);

  const [showLevelModal, setShowLevelModal] = useState(true);

  const [activeTab, setActiveTab] = useState("chat");

  const messagesEndRef = useRef(null);

  const handleSelectLevel = (chosenLevel) => {

    setLevel(chosenLevel);

    setShowLevelModal(false);

    const initialGreeting = {

      sender: "partner",

      text: `Hey there! I'm your English speaking practice partner. Since you're at the ${chosenLevel} level, we'll keep our chat tailored just for you. What's on your mind today? ❤️`,

      correction: null,

    };

    setMessages([initialGreeting]);

  };

  const scrollToBottom = () => {

    messagesEndRef.current?.scrollIntoView({ behavior: "smooth" });

  };

  useEffect(() => {

    scrollToBottom();

  }, [messages, isTyping]);

  const callClaudeAPI = async (userMessage, history, currentLevel) => {

    const systemPrompt = `You are a friendly, patient, and encouraging English speaking practice partner.

The user's level is ${currentLevel}.

- Keep responses under 4 short sentences to mimic spoken dialogue.

- Use everyday expressions and contractions naturally.

- If the user makes a grammar, vocabulary, or pronunciation mistake, include a correction block at the very end of your response using this exact format:

CORRECTION_ORIGINAL: [original]

CORRECTION_FIXED: [fixed]

CORRECTION_TIP: [short tip]

- Otherwise, just reply naturally and ask one engaging follow-up question.`;

    const apiMessages = history.map((m) => ({

      role: m.sender === "user" ? "user" : "assistant",

      content: m.text,

    }));

    apiMessages.push({ role: "user", content: userMessage });

    const controller = new AbortController();

    const timeoutId = setTimeout(() => controller.abort(), 25000);

    try {

      const response = await fetch("https://api.anthropic.com/v1/messages", {

        method: "POST",

        headers: { "Content-Type": "application/json" },

        body: JSON.stringify({

          model: "claude-sonnet-4-6",

          max_tokens: 1000,

          system: systemPrompt,

          messages: apiMessages,

        }),

        signal: controller.signal,

      });

      clearTimeout(timeoutId);

      if (!response.ok) {

        console.error("Claude API error status:", response.status);

        return "Hmm, I didn't quite catch a response there. Could you try sending that again?";

      }

      const data = await response.json();

      const textBlock = data.content?.find((b) => b.type === "text");

      return textBlock?.text || "That's interesting! Tell me more about that.";

    } catch (err) {

      clearTimeout(timeoutId);

      console.error("Claude API call failed:", err);

      return "I'm having a little trouble connecting right now, but I'm listening! What were you saying?";

    }

  };

  const isGreeting = (text) => {

    const normalized = text.toLowerCase().replace(/[^a-z\s]/g, "").trim();

    const greetings = ["hi", "hello", "hey", "hlw", "helo", "hii", "hiii"];

    return greetings.includes(normalized);

  };

  const handleSendMessage = async (e) => {

    if (e && e.preventDefault) e.preventDefault();

    const trimmed = inputText.trim();

    if (!trimmed || isTyping) return;

    const userMsg = trimmed;

    setInputText("");

    const newMessages = [...messages, { sender: "user", text: userMsg, correction: null }];

    setMessages(newMessages);

    if (isGreeting(userMsg)) {

      setIsTyping(true);

      setTimeout(() => {

        setIsTyping(false);

        setMessages((prev) => [

          ...prev,

          {

            sender: "partner",

            text: "Hi, I'm Samim. It's so nice to meet you. What's your name?",

            correction: null,

          },

        ]);

      }, 500);

      return;

    }

    setIsTyping(true);

    let rawReply;

    try {

      rawReply = await callClaudeAPI(userMsg, messages, level);

    } catch (err) {

      rawReply = "I'm having a little trouble connecting right now, but I'm listening! What were you saying?";

    }

    setIsTyping(false);

    let cleanReply = rawReply || "That's interesting! Tell me more about that.";

    let correction = null;

    // Defensively parse an optional correction block. Any malformed/partial

    // block is ignored rather than crashing the send (which previously made

    // the button look broken on some replies).

    try {

      if (cleanReply.includes("CORRECTION_ORIGINAL:")) {

        const parts = cleanReply.split("CORRECTION_ORIGINAL:");

        const head = parts[0].trim();

        const rest = parts.slice(1).join("CORRECTION_ORIGINAL:");

        if (rest.includes("CORRECTION_FIXED:")) {

          const fixedSplit = rest.split("CORRECTION_FIXED:");

          const orig = fixedSplit[0].trim();

          const tail = fixedSplit.slice(1).join("CORRECTION_FIXED:");

          let fixed = tail.trim();

          let tip = "";

          if (tail.includes("CORRECTION_TIP:")) {

            const tipSplit = tail.split("CORRECTION_TIP:");

            fixed = tipSplit[0].trim();

            tip = tipSplit.slice(1).join("CORRECTION_TIP:").trim();

          }

          if (orig && fixed) {

            cleanReply = head;

            correction = { original: orig, corrected: fixed, tip: tip };

          }

        } else {

          // Incomplete block from the model — just strip it out, no crash.

          cleanReply = head;

        }

      }

    } catch (parseErr) {

      // If anything about the correction format is unexpected, fall back

      // to showing the raw reply rather than losing the message entirely.

      cleanReply = rawReply;

      correction = null;

    }

    setMessages((prev) => [

      ...prev,

      {

        sender: "partner",

        text: cleanReply,

        correction: correction,

      },

    ]);

  };

  const triggerSpecialCommand = (cmdText) => {

    setInputText(cmdText);

  };

  return (

    <div className="flex h-screen bg-black text-white font-sans overflow-hidden select-none">

      {showLevelModal && (

        <div className="fixed inset-0 z-50 bg-black bg-opacity-80 backdrop-blur-xl flex items-center justify-center p-4">

          <div className="bg-black border border-neutral-800 rounded-3xl shadow-2xl max-w-md w-full p-8 text-center">

            <div className="w-16 h-16 bg-gradient-to-tr from-pink-500 to-rose-500 rounded-2xl flex items-center justify-center mx-auto mb-5 shadow-lg text-white">

              <Disc className="w-8 h-8 animate-spin" style={{ animationDuration: "8s" }} />

            </div>

            <h2 className="text-2xl font-semibold tracking-tight text-white mb-2">English Partner Station</h2>

            <p className="text-neutral-400 text-sm mb-8">

              Select your fluency level to personalize your vocal rhythm and conversational style.

            </p>

            <div className="space-y-3">

              {[

                { title: "Beginner", subtitle: "A1-A2 · Gentle pace & simple words", level: "Beginner" },

                { title: "Intermediate", subtitle: "B1-B2 · Natural everyday flow", level: "Intermediate" },

                { title: "Advanced", subtitle: "C1-C2 · Idioms & nuanced speech", level: "Advanced" },

              ].map((item) => (

                <button

                  key={item.level}

                  onClick={() => handleSelectLevel(item.level)}

                  className="w-full py-4 px-5 bg-neutral-900 hover:bg-neutral-800 border border-neutral-800 rounded-2xl transition flex items-center justify-between text-left group"

                >

                  <div>

                    <div className="font-medium text-white group-hover:text-rose-400 transition">{item.title}</div>

                    <div className="text-xs text-neutral-400">{item.subtitle}</div>

                  </div>

                  <ChevronRight className="w-5 h-5 text-neutral-500 group-hover:text-white transition" />

                </button>

              ))}

            </div>

          </div>

        </div>

      )}

      <div className="w-64 bg-black border-r border-neutral-800 hidden md:flex flex-col justify-between p-6">

        <div>

          <div className="flex items-center gap-3 mb-8">

            <div className="w-9 h-9 rounded-xl bg-gradient-to-tr from-rose-500 to-orange-400 flex items-center justify-center shadow-md">

              <Radio className="w-5 h-5 text-white" />

            </div>

            <span className="font-semibold tracking-tight text-white">Fluento</span>

          </div>

          <div className="text-xs font-semibold text-neutral-500 uppercase tracking-wider mb-3 px-2">Discover</div>

          <div className="space-y-1">

            <button

              onClick={() => setActiveTab("chat")}

              className={`w-full flex items-center gap-3 px-3 py-2.5 rounded-xl text-sm font-medium transition ${

                activeTab === "chat"

                  ? "bg-rose-950 text-rose-400 border border-rose-800"

                  : "text-neutral-300 hover:bg-neutral-800 hover:text-white"

              }`}

            >

              <MessageSquare className="w-4 h-4" />

              Live Conversation

            </button>

            <button

              onClick={() => setActiveTab("discover")}

              className={`w-full flex items-center gap-3 px-3 py-2.5 rounded-xl text-sm font-medium transition ${

                activeTab === "discover"

                  ? "bg-rose-950 text-rose-400 border border-rose-800"

                  : "text-neutral-300 hover:bg-neutral-800 hover:text-white"

              }`}

            >

              <Sparkles className="w-4 h-4" />

              Scenarios & Prompts

            </button>

          </div>

        </div>

        <div className="bg-neutral-900 border border-neutral-800 rounded-2xl p-4 flex items-center justify-between">

          <div>

            <div className="text-xs text-neutral-500">Current Level</div>

            <div className="font-medium text-sm text-rose-400">{level || "Not set"}</div>

          </div>

          <button

            onClick={() => setShowLevelModal(true)}

            className="text-xs bg-neutral-800 hover:bg-neutral-700 text-white px-2.5 py-1.5 rounded-lg transition"

          >

            Switch

          </button>

        </div>

      </div>

      <div className="flex-1 flex flex-col bg-black relative overflow-hidden">

        <div className="absolute top-0 left-0 right-0 h-16 bg-black border-b border-neutral-800 z-20 flex items-center justify-between px-6">

          <div className="flex items-center gap-3">

            <div className="w-3 h-3 rounded-full bg-rose-500 animate-pulse"></div>

            <span className="font-medium text-sm tracking-tight text-white">Live Practice Session</span>

          </div>

          <div className="flex items-center gap-2">

            <span className="text-xs px-2.5 py-1 rounded-full bg-neutral-900 text-neutral-200 border border-neutral-800">

              {level} Mode

            </span>

          </div>

        </div>

        {activeTab === "chat" ? (

          <div className="flex-1 overflow-y-auto pt-24 pb-32 px-4 md:px-8 space-y-6 max-w-3xl w-full mx-auto">

            {messages.map((msg, idx) => (

              <div key={idx} className={`flex flex-col ${msg.sender === "user" ? "items-end" : "items-start"}`}>

                <div

                  className={`max-w-xl rounded-2xl px-4 py-3 text-sm md:text-base leading-relaxed ${

                    msg.sender === "user"

                      ? "bg-rose-500 text-white rounded-br-sm shadow-lg"

                      : "bg-neutral-900 text-neutral-100 rounded-bl-sm border border-neutral-800"

                  }`}

                >

                  {msg.text}

                </div>

                {msg.correction && (

                  <div className="mt-2.5 max-w-xl w-full bg-neutral-900 border border-rose-900 rounded-2xl p-4 text-xs md:text-sm text-neutral-100 shadow-lg">

                    <div className="font-semibold text-rose-400 flex items-center gap-1.5 mb-1.5">

                      <span>🔧 Gentle Correction</span>

                    </div>

                    <div className="text-neutral-500 line-through mb-1">"{msg.correction.original}"</div>

                    <div className="text-emerald-400 font-medium mb-2">→ "{msg.correction.corrected}"</div>

                    {msg.correction.tip && (

                      <div className="text-neutral-300 text-xs bg-black p-2.5 rounded-xl border border-neutral-800 flex items-start gap-1.5">

                        <span>💡</span> <span>{msg.correction.tip}</span>

                      </div>

                    )}

                  </div>

                )}

              </div>

            ))}

            {isTyping && (

              <div className="flex items-center space-x-2 text-neutral-500 text-sm py-2">

                <div className="w-2 h-2 bg-rose-500 rounded-full animate-bounce"></div>

                <div className="w-2 h-2 bg-rose-500 rounded-full animate-bounce" style={{ animationDelay: "0.2s" }}></div>

                <div className="w-2 h-2 bg-rose-500 rounded-full animate-bounce" style={{ animationDelay: "0.4s" }}></div>

                <span className="text-xs ml-1">Partner is typing...</span>

              </div>

            )}

            <div ref={messagesEndRef} />

          </div>

        ) : (

          <div className="flex-1 overflow-y-auto pt-24 pb-32 px-6 max-w-3xl w-full mx-auto space-y-6">

            <h2 className="text-2xl font-bold text-white tracking-tight">Scenarios & Quick Prompts</h2>

            <p className="text-neutral-400 text-sm">

              Select any immersive situation or grammar tool to boost your fluency instantly.

            </p>

            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">

              {[

                { title: "Ordering Coffee", desc: "Practice ordering your morning latte and pastries.", cmd: "Let's practice ordering coffee" },

                { title: "Job Interview", desc: "Answer common HR questions professionally.", cmd: "Let's practice a job interview" },

                { title: "Better Phrasing", desc: "Learn how to rewrite your sentence like a native.", cmd: "Tell me a better way to say that" },

                { title: "Grammar Deep Dive", desc: "Pause and ask for clear rule breakdowns.", cmd: "Explain this" },

              ].map((card, i) => (

                <div

                  key={i}

                  onClick={() => {

                    triggerSpecialCommand(card.cmd);

                    setActiveTab("chat");

                  }}

                  className="bg-neutral-900 hover:bg-neutral-800 border border-neutral-800 rounded-2xl p-5 cursor-pointer transition group"

                >

                  <h3 className="font-semibold text-white group-hover:text-rose-400 transition mb-1">{card.title}</h3>

                  <p className="text-xs text-neutral-400 mb-4">{card.desc}</p>

                  <span className="text-xs text-rose-400 font-medium flex items-center gap-1">

                    Start now <ChevronRight className="w-3 h-3" />

                  </span>

                </div>

              ))}

            </div>

          </div>

        )}

        <div className="absolute bottom-0 left-0 right-0 bg-black border-t border-neutral-800 p-4 z-20">

          <div className="max-w-3xl mx-auto flex items-center gap-3">

            <input

              type="text"

              value={inputText}

              onChange={(e) => setInputText(e.target.value)}

              onKeyDown={(e) => {

                if (e.key === "Enter") {

                  handleSendMessage(e);

                }

              }}

              placeholder="Type your response or ask a question..."

              disabled={isTyping}

              className="flex-1 bg-neutral-900 border border-neutral-800 rounded-2xl px-5 py-3.5 text-sm text-white placeholder-neutral-500 focus:outline-none focus:ring-2 focus:ring-rose-500 transition disabled:opacity-60"

            />

            <button

              type="button"

              onClick={handleSendMessage}

              disabled={!inputText.trim() || isTyping}

              className="bg-rose-500 hover:bg-rose-600 disabled:opacity-40 text-white p-3.5 rounded-2xl transition shadow-lg flex items-center justify-center w-14 h-14"

            >

              {isTyping ? (

                <div className="w-4 h-4 border-2 border-white border-t-transparent rounded-full animate-spin"></div>

              ) : (

                <Send className="w-5 h-5" />

              )}

            </button>

          </div>

        </div>

      </div>

    </div>

  );

}
